\documentclass[border=0.05cm]{standalone}
\usepackage{tikz}
\usepackage{xcolor}
\usetikzlibrary{automata, arrows.meta, positioning}
\usepackage{fontspec}
\setmainfont{Arial}

% Define colors based on split-complementary scheme
% https://copic.too.com/blogs/educational/analogous-complimentary-and-split-complementary-color-schemes?srsltid=AfmBOoqZ8Nt71usioxMxPyKzjQ5vDcxs2B_imasd_wZa50UB5cZX6xCL
\definecolor{dftcolor}{RGB}{44, 163, 207}  
\definecolor{simcolor}{RGB}{237, 156, 44} 
\definecolor{expcolor}{RGB}{222, 70, 28}

% Define styles
\tikzset{
	dft/.style={draw=dftcolor!80!black, fill=dftcolor!40, minimum size=2cm},
	simulated/.style={draw=simcolor!80!black, fill=simcolor!30, minimum size=2cm},
	experimental/.style={draw=expcolor!80!black, fill=expcolor!40, minimum size=2cm},
	YXP/.style={thick, ->, color=simcolor},
	XPY/.style={thick, ->, color=dftcolor},
	XPX/.style={thick, ->, color=expcolor},
	XXP/.style={thick, ->, color=simcolor},
	XY/.style={thick, ->, color=dftcolor},
	annotation/.style={font=\small}
}

\begin{document}
	
	\begin{tikzpicture}[node distance = 5cm, on grid, auto]
		
		% Nodes
		\node (Y) [state, initial, accepting, initial text = {DFT}, dft] {$\mathcal{Y}$};
		\node (X') [state, above right = of Y, simulated] {$\mathcal{X}^{\prime}$};
		\node (X) [state, below right = of X', experimental] {$\mathcal{X}$};
		
		% Annotations
		\node[below=1.2cm of Y, annotation, text=dftcolor!80!black] {Atomic configuration};
		\node[below=0.6cm of Y, annotation] {graphs};
		
		\node[above=1.2cm of X', annotation, text=simcolor!80!black] {Simulated AFM};
		\node[below=0.6cm of X', annotation] {images};
		
		\node[below=1.2cm of X, annotation, text=expcolor!90!black] {Experimental AFM};
		\node[below=0.6cm of X, annotation] {images};
		
		% Arrows (transitions)
		\path [YXP] 
		(Y)  edge [bend left] node[sloped, above] {$F_{\mathcal{Y}}$} (X');
		\path [XPY]
		(X') edge [bend left]   node[sloped, above]    {$S_{\mathcal{X}^{\prime}}$}    (Y);
		\path [XPX]
		(X') edge [bend left]   node[sloped, above]    {$G_{\mathcal{X}^{\prime}}$}    (X);
		\path [XXP]
		(X)  edge [bend left]   node[sloped, above, yshift=2pt]    {$G_{\mathcal{X}}$}   (X');
		\path [XY]
		(X)  edge [bend left]   node[sloped, above]    {$S_{\mathcal{X}}$}    (Y);
		
	\end{tikzpicture}
	
\end{document}
